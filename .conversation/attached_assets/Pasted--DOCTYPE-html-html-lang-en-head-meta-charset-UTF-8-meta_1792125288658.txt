<script src="https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.min.js"></script>
<script>
const $=id=>document.getElementById(id);
const colors=[0xe53935,0x1e88e5,0xfdd835,0x43a047,0xffffff,0x8e24aa];
let selected=0,scene,camera,renderer,player;
let traffic=[],coins=[],marks=[],trees=[],tolls=[];
let running=false,paused=false;
let speed=0,distance=0,coinCount=0,time=0;
let spawn=0,coinSpawn=0,tollSpawn=0;
let targetX=0,left=false,right=false,gas=false,brake=false;
let mode='endless';
const clock=new THREE.Clock();
const lanes=[-3,0,3];

colors.forEach((c,i)=>{
 const b=document.createElement('button');
 b.className='carPick'+(i===0?' selected':'');
 b.textContent=['🔴 Red','🔵 Blue','🟡 Turbo','🟢 Green','⚪ White','🟣 Purple'][i];
 b.onclick=()=>{
  selected=i;
  [...$('garage').children].forEach((x,j)=>x.classList.toggle('selected',j===i));
 };
 $('garage').appendChild(b);
});

function carModel(color,scale=1){
 const g=new THREE.Group();
 const paint=new THREE.MeshStandardMaterial({color,metalness:.25,roughness:.35});
 function box(w,h,d,x,y,z,mat){
  const m=new THREE.Mesh(new THREE.BoxGeometry(w,h,d),mat);
  m.position.set(x,y,z);m.castShadow=true;g.add(m);return m;
 }
 box(1.7,.55,3.1,0,.65,0,paint);
 box(1.25,.62,1.4,0,1.15,.1,
  new THREE.MeshStandardMaterial({color:0x172b3b,metalness:.3}));
 for(const x of [-.85,.85])for(const z of [-1,1]){
  const w=new THREE.Mesh(new THREE.CylinderGeometry(.3,.3,.2,12),
   new THREE.MeshStandardMaterial({color:0x111111}));
  w.rotation.z=Math.PI/2;w.position.set(x,.35,z);g.add(w);
 }
 for(const x of [-.55,.55]){
  box(.3,.14,.06,x,.68,-1.56,
   new THREE.MeshStandardMaterial({color:0xffffcc,emissive:0x444400}));
  box(.28,.14,.06,x,.65,1.56,
   new THREE.MeshStandardMaterial({color:0xff2222,emissive:0x550000}));
 }
 g.scale.setScalar(scale);return g;
}

function init(){
 scene=new THREE.Scene();
 scene.background=new THREE.Color(0x87ceeb);
 scene.fog=new THREE.Fog(0x87ceeb,45,190);
 camera=new THREE.PerspectiveCamera(65,innerWidth/innerHeight,.1,350);
 camera.position.set(0,6.5,10);
 renderer=new THREE.WebGLRenderer({antialias:true});
 renderer.setSize(innerWidth,innerHeight);
 renderer.setPixelRatio(Math.min(devicePixelRatio,1.5));
 $('game').appendChild(renderer.domElement);
 scene.add(new THREE.HemisphereLight(0xffffff,0x52654b,2));
 const sun=new THREE.DirectionalLight(0xffffff,2);
 sun.position.set(-8,16,8);scene.add(sun);

 const ground=new THREE.Mesh(new THREE.PlaneGeometry(400,500),
  new THREE.MeshStandardMaterial({color:0x43884a}));
 ground.rotation.x=-Math.PI/2;
 ground.position.set(0,-.15,-120);scene.add(ground);

 const road=new THREE.Mesh(new THREE.PlaneGeometry(12,500),
  new THREE.MeshStandardMaterial({color:0x30343a}));
 road.rotation.x=-Math.PI/2;
 road.position.set(0,-.05,-120);scene.add(road);

 for(const x of [-6.2,6.2]){
  const curb=new THREE.Mesh(new THREE.BoxGeometry(.4,.12,500),
   new THREE.MeshStandardMaterial({color:0xffffff}));
  curb.position.set(x,0,-120);scene.add(curb);
 }
 for(let z=-350;z<20;z+=12){
  for(const x of [-1.5,1.5]){
   const m=new THREE.Mesh(new THREE.BoxGeometry(.12,.025,5),
    new THREE.MeshStandardMaterial({color:0xffffff}));
   m.position.set(x,.01,z);scene.add(m);marks.push(m);
  }
 }
 for(let i=0;i<42;i++){
  for(const side of [-1,1]){
   const g=new THREE.Group();
   const trunk=new THREE.Mesh(new THREE.CylinderGeometry(.18,.25,1.5,7),
    new THREE.MeshStandardMaterial({color:0x76502e}));
   trunk.position.y=.75;g.add(trunk);
   const leaves=new THREE.Mesh(new THREE.ConeGeometry(1.1,2.8,7),
    new THREE.MeshStandardMaterial({color:0x287b42}));
   leaves.position.y=2.5;g.add(leaves);
   g.position.set(side*(10+Math.random()*5),0,-i*10-10);
   scene.add(g);trees.push(g);
  }
 }
 makePlayer();
 addEventListener('resize',()=>{
  camera.aspect=innerWidth/innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth,innerHeight);
 });
}
function makePlayer(){
 if(player)scene.remove(player);
 player=carModel(colors[selected]);
 player.position.set(0,0,3);scene.add(player);
}
function clear(arr){
 arr.forEach(o=>scene.remove(o));arr.length=0;
}
function addCoin(x,z){
 const g=new THREE.Group();
 const gold=new THREE.MeshStandardMaterial({
  color:0xffc928,metalness:.6,emissive:0x553300
 });
 const ring=new THREE.Mesh(new THREE.TorusGeometry(.4,.12,8,18),gold);
 ring.rotation.y=Math.PI/2;g.add(ring);
 g.position.set(x,1.2,z);scene.add(g);coins.push(g);
}
function addTraffic(){
 const lane=Math.floor(Math.random()*3);
 const car=carModel(colors[Math.floor(Math.random()*colors.length)]);
 car.position.set(lanes[lane],0,-145);
 car.userData.truck=Math.random()<.15;
 if(car.userData.truck){
  const box=new THREE.Mesh(new THREE.BoxGeometry(1.7,1,1.5),
   new THREE.MeshStandardMaterial({color:0xdddddd}));
  box.position.y=1.3;car.add(box);
 }
 scene.add(car);traffic.push(car);
}
function addToll(){
 const g=new THREE.Group();
 g.position.z=-170;
 const mat=new THREE.MeshStandardMaterial({color:0x0f766e});
 for(const x of [-4.5,0,4.5]){
  const post=new THREE.Mesh(new THREE.BoxGeometry(.35,3.5,.4),mat);
  post.position.set(x,1.75,0);g.add(post);
  const sign=new THREE.Mesh(new THREE.BoxGeometry(2,.6,.2),
   new THREE.MeshStandardMaterial({color:0xfacc15}));
  sign.position.set(x,2.8,0);g.add(sign);
 }
 const roof=new THREE.Mesh(new THREE.BoxGeometry(11,.4,1.3),mat);
 roof.position.y=3.5;g.add(roof);
 scene.add(g);tolls.push(g);
}
function start(){
 mode=$('mode').value;
 makePlayer();clear(traffic);clear(coins);clear(tolls);
 speed=0;distance=0;coinCount=0;time=0;
 spawn=0;coinSpawn=0;tollSpawn=0;targetX=0;
 running=true;paused=false;
 $('overlay').classList.add('hidden');
 $('controls').style.display='flex';
 $('start').textContent='▶ START RACE';
 clock.getDelta();
}
function end(title){
 running=false;
 $('overlay').classList.remove('hidden');
 $('overlay').querySelector('h1').innerHTML=title;
 $('overlay').querySelector('p').textContent=
  'Distance: '+Math.floor(distance)+' m | Coins: '+coinCount;
 $('start').textContent='↻ RACE AGAIN';
 $('controls').style.display='none';
}
function update(){
 requestAnimationFrame(update);
 const dt=Math.min(clock.getDelta(),.04);
 if(running&&!paused){
  time+=dt;
  speed=gas?Math.min(13,speed+dt*5):Math.max(5,speed-dt);
  if(brake)speed=Math.max(0,speed-dt*10);
  if(left)targetX=Math.max(-3,targetX-dt*7);
  if(right)targetX=Math.min(3,targetX+dt*7);
  player.position.x+=(targetX-player.position.x)*Math.min(1,dt*8);
  const move=speed*dt*2.3;
  distance+=move*2;
  marks.forEach(m=>{
   m.position.z+=move;if(m.position.z>12)m.position.z-=360;
  });
  trees.forEach(t=>{
   t.position.z+=move;if(t.position.z>15)t.position.z-=420;
  });
  spawn+=dt;
  if(spawn>Math.max(.55,1.65-speed*.055)){addTraffic();spawn=0;}
  coinSpawn+=dt;
  if(coinSpawn>1.2){
   const x=lanes[Math.floor(Math.random()*3)];
   for(let i=0;i<4;i++)addCoin(x,-100-i*4);
   coinSpawn=0;
  }
  tollSpawn+=dt;
  if(tollSpawn>24){addToll();tollSpawn=0;}

  for(let i=traffic.length-1;i>=0;i--){
   const c=traffic[i];
   c.position.z+=move*(c.userData.truck?.68:.88);
   if(Math.abs(c.position.x-player.position.x)<1.45&&
      Math.abs(c.position.z-player.position.z)<2.2){end('💥 CRASH!');}
   if(c.position.z>18){scene.remove(c);traffic.splice(i,1);}
  }
  for(let i=coins.length-1;i>=0;i--){
   const c=coins[i];c.position.z+=move;c.rotation.y+=dt*3;
   if(Math.abs(c.position.z-player.position.z)<2&&
      Math.abs(c.position.x-player.position.x)<1.4){
    coinCount++;c.position.z=1000;
   }
   if(c.position.z>15||c.position.z>900){
    scene.remove(c);coins.splice(i,1);
   }
  }
  for(let i=tolls.length-1;i>=0;i--){
   const t=tolls[i];t.position.z+=move;
   if(t.position.z>20){scene.remove(t);tolls.splice(i,1);}
  }
  if(mode==='time'&&time>=60)end('⏱ TIME UP!');
  if(mode==='traffic'&&distance>=1000)end('🏆 YOU WIN!');
  $('coins').textContent=coinCount;
  $('distance').textContent=Math.floor(distance);
 }
 camera.position.x+=(player.position.x*.34-camera.position.x)*.04;
 camera.lookAt(player.position.x*.2,1,-13);
 renderer.render(scene,camera);
}
function hold(id,fn){
 const el=$(id);
 el.addEventListener('pointerdown',e=>{e.preventDefault();fn(true);});
 ['pointerup','pointercancel','pointerleave'].forEach(ev=>
  el.addEventListener(ev,()=>fn(false)));
}
hold('left',v=>left=v);hold('right',v=>right=v);
hold('gas',v=>gas=v);hold('brake',v=>brake=v);
addEventListener('keydown',e=>{
 if(e.key==='ArrowLeft'||e.key==='a')left=true;
 if(e.key==='ArrowRight'||e.key==='d')right=true;
 if(e.key==='ArrowUp'||e.key==='w')gas=true;
 if(e.key==='ArrowDown'||e.key==='s')brake=true;
});
addEventListener('keyup',e=>{
 if(e.key==='ArrowLeft'||e.key==='a')left=false;
 if(e.key==='ArrowRight'||e.key==='d')right=false;
 if(e.key==='ArrowUp'||e.key==='w')gas=false;
 if(e.key==='ArrowDown'||e.key==='s')brake=false;
});
$('start').onclick=()=>{
 if(running&&paused){paused=false;$('overlay').classList.add('hidden');}
 else start();
};
$('pause').onclick=()=>{
 if(!running)return;
 paused=!paused;
 $('overlay').classList.toggle('hidden',!paused);
 $('start').textContent=paused?'▶ RESUME':'▶ START RACE';
};
init();update();
</script>
